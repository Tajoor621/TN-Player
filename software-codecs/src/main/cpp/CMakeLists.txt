# TN-Player glue; upstream JNI sources retain their Apache-2.0 headers.
cmake_minimum_required(VERSION 3.22)
project(tn_codecs C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_SHARED_LINKER_FLAGS "${CMAKE_SHARED_LINKER_FLAGS} -Wl,-z,max-page-size=16384 -Wl,--gc-sections")
add_compile_options(-O2 -ffunction-sections -fdata-sections)
foreach(lib avcodec avutil swresample)
  add_library(${lib} STATIC IMPORTED)
  set_target_properties(${lib} PROPERTIES IMPORTED_LOCATION "${CODEC_PREFIX}/lib/lib${lib}.a")
endforeach()
add_library(ffmpegJNI SHARED ffmpeg/ffmpeg_jni.cc)
target_include_directories(ffmpegJNI PRIVATE "${CODEC_PREFIX}/include")
target_link_libraries(ffmpegJNI swresample avcodec avutil android log m)
# Bind FFmpeg AArch64 assembly table references locally (same as upstream module).
target_link_options(ffmpegJNI PRIVATE "-Wl,-Bsymbolic")
add_library(vpx STATIC IMPORTED)
set_target_properties(vpx PROPERTIES IMPORTED_LOCATION "${CODEC_PREFIX}/lib/libvpx.a")
add_library(ndk_cpu STATIC "${ANDROID_NDK}/sources/android/cpufeatures/cpu-features.c")
target_include_directories(ndk_cpu PUBLIC "${ANDROID_NDK}/sources/android/cpufeatures")
add_library(vpxV2JNI SHARED vp9/vpx_jni.cc)
target_include_directories(vpxV2JNI PRIVATE "${CODEC_PREFIX}/include")
target_link_libraries(vpxV2JNI vpx ndk_cpu android log m z)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_EXECUTABLE OFF CACHE BOOL "" FORCE)
set(LIBGAV1_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(LIBGAV1_ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LIBGAV1_THREADPOOL_USE_STD_MUTEX 1 CACHE STRING "" FORCE)
add_subdirectory("${CODEC_SOURCES}/cpu_features" cpu_features EXCLUDE_FROM_ALL)
add_subdirectory("${CODEC_SOURCES}/libgav1" libgav1 EXCLUDE_FROM_ALL)
add_library(gav1JNI SHARED av1/gav1_jni.cc av1/cpu_info.cc)
target_link_libraries(gav1JNI android cpu_features libgav1_static log)
if(ANDROID_ABI STREQUAL "armeabi-v7a")
  target_compile_options(gav1JNI PRIVATE -mfpu=neon -marm)
endif()
